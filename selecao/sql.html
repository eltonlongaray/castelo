SQL da tabela da seleção

  1. Abre o Supabase → projeto realeza-holistica (precisa estar ativo, não "Coming up").
  2. SQL Editor → New query.
  3. Toca em Copiar SQL aqui embaixo, cola lá e clica em Run.
  4. Tem que aparecer "Success. No rows returned". Rode só uma vez.

create table if not exists public.selecao_aula (
  id bigint generated always as identity primary key,
  criado_em timestamptz not null default now(),
  nome text, instagram text, whatsapp text, cidade text,
  tempo_atende text, atend_semana text, renda text, clientes_voltam text, pacotes text,
  origem_clientes text, maior_desafio text, como_se_sente text, nao_deveria text,
  o_que_falta text, ja_tentou text, rotina_ideal text,
  por_que_vaga text, como_ajuda text,
  selecionada boolean not null default false
);
alter table public.selecao_aula enable row level security;

create policy "qualquer pessoa envia" on public.selecao_aula
  for insert to anon, authenticated with check (selecionada = false);
create policy "admin le" on public.selecao_aula
  for select to authenticated using (auth.jwt()->>'email' = 'elton.longaray483@gmail.com');
create policy "admin seleciona" on public.selecao_aula
  for update to authenticated
  using (auth.jwt()->>'email' = 'elton.longaray483@gmail.com')
  with check (auth.jwt()->>'email' = 'elton.longaray483@gmail.com');
